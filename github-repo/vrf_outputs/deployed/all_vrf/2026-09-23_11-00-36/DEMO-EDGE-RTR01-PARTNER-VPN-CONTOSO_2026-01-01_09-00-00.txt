vrf definition PARTNER-VPN-CONTOSO
  rd 65000:100
  !
  address-family ipv4
  exit-address-family
  !

! Interfaces
interface GigabitEthernet0/1.100
  description +++ PARTNER-VPN-CONTOSO LAN +++
  encapsulation dot1Q 100
  vrf forwarding PARTNER-VPN-CONTOSO
  ip address 192.0.2.1 255.255.255.0
  ip nat inside
  !
  interface Tunnel100
    description *** IPSec VTI to Contoso (demo) ***
    vrf forwarding PARTNER-VPN-CONTOSO
    ip unnumbered Loopback0
    ip nat outside
    ip tcp adjust-mss 1380
    tunnel source GigabitEthernet0/0
    tunnel mode ipsec ipv4
    tunnel destination 198.51.100.2
    tunnel protection ipsec profile CONTOSO-IPSEC-PROFILE
    !

router bgp 65000
